# Docs build dependencies.
#
# Both workflows that build the docs (docs-preview.yml on PRs, deploy-site.yml
# on main) install from this file, so a plugin added here reaches production and
# the PR gate together. They used to `pip install mkdocs-material` inline, which
# meant adding a plugin in one place silently broke the build in the other.
#
# Minor versions are pinned loosely: mkdocs-material ships breaking template
# changes in majors, and static-i18n follows mkdocs' plugin API closely.
mkdocs-material>=9.7,<10
mkdocs-static-i18n>=1.3,<2
